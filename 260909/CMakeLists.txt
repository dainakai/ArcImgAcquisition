cmake_minimum_required(VERSION 3.24)
project(dual_holo VERSION 1.2.4 LANGUAGES C CXX)
if(APPLE)
  # Enable before adding OpenCV: its Cocoa sources also need this language's
  # compile rules in the parent scope during CMake generation.
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "DualHolo requires a 64-bit toolchain (Windows/Linux x64 or macOS arm64/x64)")
endif()
# Static CRT allows the Windows app to run without a separate VC redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HOLO_BUNDLED_OPENCV "Build a minimal static OpenCV for distributable packages" OFF)
if(HOLO_BUNDLED_OPENCV)
  include(cmake/OpenCV.cmake)
else()
  if(APPLE AND CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64" AND NOT DEFINED OpenCV_DIR)
    list(PREPEND CMAKE_PREFIX_PATH /opt/homebrew/opt/opencv)
  endif()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
endif()
find_package(Threads REQUIRED)
# OpenCV sets the legacy EXECUTABLE_OUTPUT_PATH in the cache; keep our layout stable.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

function(holo_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

add_library(holo_core STATIC src/core.cpp src/recorder.cpp)
target_include_directories(holo_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(holo_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
holo_warnings(holo_core)
add_library(holo_runtime STATIC src/spin_api.cpp src/camera_lock.cpp)
target_include_directories(holo_runtime PUBLIC src)
target_link_libraries(holo_runtime PUBLIC ${CMAKE_DL_LIBS})
holo_warnings(holo_runtime)
add_executable(dual_holo src/main.cpp src/cameras.cpp)
target_link_libraries(dual_holo PRIVATE holo_core holo_runtime)
target_compile_definitions(dual_holo PRIVATE HOLO_VERSION="${PROJECT_VERSION}")
holo_warnings(dual_holo)
if(WIN32)
  target_sources(dual_holo PRIVATE src/windows_app.cpp)
  target_link_libraries(dual_holo PRIVATE user32 shell32 ole32 uuid)
endif()
if(APPLE)
  set_source_files_properties(config.yml PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(dual_holo PRIVATE src/mac_app.mm config.yml)
  target_link_libraries(dual_holo PRIVATE "-framework AppKit")
  set_target_properties(dual_holo PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME DualHolo
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/Info.plist")
  set_target_properties(dual_holo PROPERTIES OUTPUT_NAME DualHolo)
endif()
add_executable(inspect_cameras src/inspect.cpp)
target_link_libraries(inspect_cameras PRIVATE holo_runtime)
holo_warnings(inspect_cameras)
if(APPLE)
  # The vendor's C library references GenICam dylibs via @rpath.
  set_target_properties(dual_holo inspect_cameras PROPERTIES
    BUILD_RPATH "/Applications/Spinnaker/lib;/usr/local/lib"
    INSTALL_RPATH "/Applications/Spinnaker/lib;/usr/local/lib")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(holo_tests tests/test_core.cpp)
  target_link_libraries(holo_tests PRIVATE holo_core)
  add_test(NAME holo_tests COMMAND holo_tests)
  add_executable(lock_tests tests/test_lock.cpp)
  target_link_libraries(lock_tests PRIVATE holo_runtime)
  add_test(NAME lock_tests COMMAND lock_tests)
  add_library(fake_spinnaker SHARED tests/fake_spinnaker.cpp)
  target_include_directories(fake_spinnaker PRIVATE src)
  add_executable(camera_tests tests/test_cameras.cpp src/cameras.cpp)
  target_link_libraries(camera_tests PRIVATE holo_core holo_runtime)
  holo_warnings(camera_tests)
  foreach(mode RANGE 0 2)
    add_test(NAME camera_runtime_${mode} COMMAND ${CMAKE_COMMAND} -E env
      "SPINNAKER_C_LIBRARY=$<TARGET_FILE:fake_spinnaker>" "HOLO_FAKE_FORMAT=${mode}" "$<TARGET_FILE:camera_tests>")
    set_tests_properties(camera_runtime_${mode} PROPERTIES RESOURCE_LOCK cameras TIMEOUT 30)
  endforeach()
  add_test(NAME camera_init_failure COMMAND ${CMAKE_COMMAND} -E env
    "SPINNAKER_C_LIBRARY=$<TARGET_FILE:fake_spinnaker>" "HOLO_FAKE_FORMAT=0" "HOLO_FAKE_INIT_FAILURE=1" "$<TARGET_FILE:camera_tests>")
  set_tests_properties(camera_init_failure PROPERTIES RESOURCE_LOCK cameras TIMEOUT 30)
  set(HOLO_SPINNAKER_INCLUDE_DIR "" CACHE PATH "Optional Spinnaker include/spinc path for ABI verification")
  if(HOLO_SPINNAKER_INCLUDE_DIR)
    add_executable(sdk_abi_tests tests/test_sdk_abi.cpp)
    target_include_directories(sdk_abi_tests PRIVATE src "${HOLO_SPINNAKER_INCLUDE_DIR}")
    add_test(NAME sdk_abi_tests COMMAND sdk_abi_tests)
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(WIN32)
    add_test(NAME windows_runtime_discovery COMMAND ${Python3_EXECUTABLE}
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_windows_runtime.py"
      "$<TARGET_FILE:inspect_cameras>" "$<TARGET_FILE:fake_spinnaker>")
    set_tests_properties(windows_runtime_discovery PROPERTIES TIMEOUT 30)
  endif()
  add_test(NAME simulation COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:dual_holo>")
  set_tests_properties(holo_tests lock_tests simulation PROPERTIES TIMEOUT 90)
endif()

# A release is only portable when OpenCV is built into the executables.
if(HOLO_BUNDLED_OPENCV)
  install(TARGETS dual_holo inspect_cameras
    RUNTIME DESTINATION . COMPONENT app
    BUNDLE DESTINATION . COMPONENT app)
  install(FILES config.yml README.md docs/distribution.md DESTINATION . COMPONENT app)
  install(DIRECTORY "${CMAKE_BINARY_DIR}/third-party-licenses/" DESTINATION licenses COMPONENT app)
  if(APPLE)
    install(PROGRAMS scripts/Simulate.command DESTINATION . COMPONENT app)
  elseif(WIN32)
    install(FILES scripts/Simulate.cmd scripts/Run.cmd DESTINATION . COMPONENT app)
  else()
    install(PROGRAMS scripts/run.sh DESTINATION . COMPONENT app)
  endif()
  set(CPACK_PACKAGE_NAME DualHolo)
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  set(CPACK_PACKAGE_VENDOR dainakai)
  set(HOLO_PACKAGE_PLATFORM "${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING "Release platform label")
  set(CPACK_PACKAGE_FILE_NAME "DualHolo-${PROJECT_VERSION}-${HOLO_PACKAGE_PLATFORM}")
  set(CPACK_GENERATOR ZIP)
  if(UNIX AND NOT APPLE)
    set(CPACK_GENERATOR TGZ)
  endif()
  set(CPACK_INSTALL_CMAKE_PROJECTS "${CMAKE_BINARY_DIR};${PROJECT_NAME};app;/")
  include(CPack)
endif()

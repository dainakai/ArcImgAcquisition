cmake_minimum_required(VERSION 3.20)
project(dual_holo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
list(PREPEND CMAKE_PREFIX_PATH /opt/homebrew/opt/opencv)
# The machine also has an Intel OpenCV under /usr/local. Prefer native Homebrew.
if(APPLE AND CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64" AND NOT DEFINED OpenCV_DIR)
  file(GLOB OPENCV_CONFIG_DIRS /opt/homebrew/opt/opencv/lib/cmake/opencv*)
  if(OPENCV_CONFIG_DIRS)
    list(GET OPENCV_CONFIG_DIRS 0 OpenCV_DIR)
  endif()
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
find_package(Threads REQUIRED)
set(SPINNAKER_ROOT /Applications/Spinnaker CACHE PATH "Spinnaker SDK installation")
find_path(SPINNAKER_INCLUDE Spinnaker.h PATHS ${SPINNAKER_ROOT}/include /opt/spinnaker/include REQUIRED)
find_library(SPINNAKER_LIB Spinnaker PATHS ${SPINNAKER_ROOT}/lib /opt/spinnaker/lib REQUIRED)
add_library(holo_core src/core.cpp src/recorder.cpp)
target_include_directories(holo_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(holo_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
target_compile_options(holo_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dual_holo src/main.cpp src/cameras.cpp)
target_include_directories(dual_holo SYSTEM PRIVATE ${SPINNAKER_INCLUDE})
target_link_libraries(dual_holo PRIVATE holo_core ${SPINNAKER_LIB})
target_compile_options(dual_holo PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(dual_holo PROPERTIES BUILD_RPATH "${SPINNAKER_ROOT}/lib;/usr/local/lib")
if(APPLE)
  add_custom_command(TARGET dual_holo POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/DualHolo.app/Contents/MacOS"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:dual_holo>" "${CMAKE_BINARY_DIR}/DualHolo.app/Contents/MacOS/dual_holo"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/src/Info.plist" "${CMAKE_BINARY_DIR}/DualHolo.app/Contents/Info.plist")
endif()
add_executable(inspect_cameras src/inspect.cpp)
target_include_directories(inspect_cameras SYSTEM PRIVATE ${SPINNAKER_INCLUDE})
target_link_libraries(inspect_cameras PRIVATE ${SPINNAKER_LIB})
set_target_properties(inspect_cameras PROPERTIES BUILD_RPATH "${SPINNAKER_ROOT}/lib;/usr/local/lib")
enable_testing()
add_executable(holo_tests tests/test_core.cpp)
target_link_libraries(holo_tests PRIVATE holo_core)
add_test(NAME holo_tests COMMAND holo_tests)
